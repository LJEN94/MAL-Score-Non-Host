// ==UserScript==
// @name         MAL Inline Scores
// @namespace    https://myanimelist.net/
// @version      4.4.0
// @description  Shows the MyAnimeList score next to every anime/manga link, plus whether it is on your own anime list; persistent shared cache, parallel rate-limited fetching, export/import and bulk preload
// @match        https://myanimelist.net/*
// @grant        GM_xmlhttpRequest
// @grant        GM_getValue
// @grant        GM_setValue
// @grant        GM_deleteValue
// @grant        GM_listValues
// @grant        GM_registerMenuCommand
// @connect      api.jikan.moe
// @run-at       document-end
// ==/UserScript==

(function () {
  'use strict';

  const PREFIX = 'mal:'; // live score cache entries
  const BULK_PREFIX = 'bulk:anime:'; // imported bulk dataset, separate from live cache
  const FRESH_MS = 7 * 24 * 60 * 60 * 1000; // scored entries are reused for 7 days
  const NO_SCORE_FRESH_MS = 6 * 60 * 60 * 1000; // entries without a score are rechecked after 6 hours
  const PRUNE_MS = 90 * 24 * 60 * 60 * 1000; // entries not refreshed for 90 days are deleted
  const LINK_SELECTOR = 'a[href*="/anime/"], a[href*="/manga/"]';
  const LINK_RE = /^\/(anime|manga)\/(\d+)(?:\/[^/?#]*)?\/?$/;

  const log = (...args) => console.log('[MAL Inline Scores]', ...args);
  const sleep = ms => new Promise(r => setTimeout(r, ms));
  log('loaded');

  // ---------- styles ----------
  const style = document.createElement('style');
  style.textContent = `
    /* One pill per link: [★ score | list status]. Declarations are !important and the
       base is reset with "all: initial" so the site's own span styles can't break it. */
    html .mal-chip.mal-chip {
      all: initial !important;
      display: inline-flex !important;
      align-items: stretch !important;
      vertical-align: middle !important;
      margin: 0 0 0 6px !important;
      border-radius: 999px !important;
      overflow: hidden !important;
      box-shadow: 0 1px 2px rgba(0, 0, 0, 0.28) !important;
      font: 600 11px/1 -apple-system, "Segoe UI", Roboto, Helvetica, Arial, sans-serif !important;
      font-variant-numeric: tabular-nums !important;
      white-space: nowrap !important;
      cursor: help !important;
      text-decoration: none !important;
    }
    .mal-chip .mal-seg {
      all: initial !important;
      display: inline-block !important;
      padding: 3px 8px !important;
      color: #fff !important;
      font: inherit !important;
      line-height: 1 !important;
      letter-spacing: 0.01em !important;
      text-decoration: none !important;
      cursor: help !important;
    }
    .mal-chip .mal-score::before {
      content: "★" !important;
      margin-right: 3px !important;
      font-size: 10px !important;
      opacity: 0.85 !important;
    }
    .mal-chip .mal-score.t-great { background: #1f9d55 !important; }
    .mal-chip .mal-score.t-good { background: #3566c9 !important; }
    .mal-chip .mal-score.t-mixed { background: #c27c0e !important; }
    .mal-chip .mal-score.t-poor { background: #c2410c !important; }
    .mal-chip .mal-score.na { background: #7b8190 !important; }
    .mal-chip .mal-score.loading {
      background: #7b8190 !important;
      animation: mal-pulse 1.2s ease-in-out infinite !important;
    }
    @keyframes mal-pulse { 0%, 100% { opacity: 0.55; } 50% { opacity: 1; } }
    .mal-chip .mal-list { border-left: 1px solid rgba(255, 255, 255, 0.45) !important; }
    .mal-chip .mal-list.s1 { background: #2db039 !important; }
    .mal-chip .mal-list.s2 { background: #26448f !important; }
    .mal-chip .mal-list.s3 { background: #f9d457 !important; color: #3a2f00 !important; }
    .mal-chip .mal-list.s4 { background: #a12f31 !important; }
    .mal-chip .mal-list.s6 { background: #c3c8d1 !important; color: #2b2f38 !important; }
    html .mal-seen.mal-seen.mal-seen { opacity: 0.4 !important; transition: opacity 0.15s !important; }
    html .mal-seen.mal-seen.mal-seen:hover { opacity: 1 !important; }
    html .mal-hide-seen.mal-hide-seen { display: none !important; }
  `;
  document.head.appendChild(style);

  // ---------- persistent cache (shared by every MAL tab) ----------
  const getEntry = key => GM_getValue(PREFIX + key, null);
  const setEntry = (key, data) => GM_setValue(PREFIX + key, { t: Date.now(), ...data });
  const isFresh = e => e && Number.isFinite(e.t) && Date.now() - e.t < (validScore(e.score) ? FRESH_MS : NO_SCORE_FRESH_MS);
  const allKeys = () => GM_listValues().filter(k => k.startsWith(PREFIX));
  const bulkKeys = () => GM_listValues().filter(k => k.startsWith(BULK_PREFIX));
  const getBulkEntry = (type, id) => type === 'anime' ? GM_getValue(BULK_PREFIX + id, null) : null;
  const validScore = value => typeof value === 'number' && Number.isFinite(value) && value > 0 && value <= 10;
  const normaliseEntry = value => ({
    score: validScore(value.score) ? value.score : null,
    scoredBy: Number.isFinite(Number(value.scoredBy ?? value.scored_by)) ? Number(value.scoredBy ?? value.scored_by) || null : null,
    rank: Number.isFinite(Number(value.rank)) ? Number(value.rank) || null : null,
    title: typeof value.title === 'string' ? value.title : '',
  });

  // Delete entries that haven't been refreshed in 90 days (runs at most once a day).
  function prune() {
    const last = GM_getValue('meta:lastPrune', 0);
    if (Date.now() - last < 24 * 60 * 60 * 1000) return;
    GM_setValue('meta:lastPrune', Date.now());
    let removed = 0;
    for (const k of allKeys()) {
      const e = GM_getValue(k);
      if (!e || Date.now() - e.t > PRUNE_MS) {
        GM_deleteValue(k);
        removed++;
      }
    }
    if (removed) log(`pruned ${removed} old cache entries`);
  }
  setTimeout(prune, 5000);

  // ---------- data source 1: Jikan API ----------
  function jikanRequest(path) {
    return new Promise(resolve => {
      GM_xmlhttpRequest({
        method: 'GET',
        url: 'https://api.jikan.moe/v4' + path,
        headers: { Accept: 'application/json' },
        timeout: 15000,
        onload: res => {
          let json = null;
          try { json = JSON.parse(res.responseText); } catch (e) { /* not JSON */ }
          resolve({ status: res.status, json });
        },
        onerror: () => resolve({ status: 0, json: null }),
        ontimeout: () => resolve({ status: 0, json: null }),
      });
    });
  }

  // For bulk/list requests: retries on rate limiting, returns parsed JSON or null.
  async function jikan(path, tries = 0) {
    const res = await jikanRequest(path);
    if ((res.status === 429 || res.status >= 500) && tries < 3) {
      await sleep(2500 * (tries + 1));
      return jikan(path, tries + 1);
    }
    return res.status === 200 ? res.json : null;
  }

  const toEntry = d => ({
    score: d.score, scoredBy: d.scored_by, rank: d.rank, title: d.title,
  });

  async function jikanEntry(type, id, lane) {
    const res = await jikanRequest(`/${type}/${id}`);
    if (res.status === 200 && res.json && res.json.data) return toEntry(res.json.data);
    if (res.status === 429 || res.status >= 500 || res.status === 0) penalize(lane, 4000);
    return null;
  }

  // ---------- data source 2: the MAL page itself (same origin) ----------
  function parseLd(doc) {
    for (const s of doc.querySelectorAll('script[type="application/ld+json"]')) {
      try {
        const agg = JSON.parse(s.textContent).aggregateRating;
        if (agg) {
          return {
            score: parseFloat(agg.ratingValue),
            scoredBy: parseInt(agg.ratingCount, 10),
          };
        }
      } catch (e) { /* ignore malformed block */ }
    }
    return null;
  }

  async function scrapeMal(type, id, lane) {
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(), 15000);
    try {
      const res = await fetch(`https://myanimelist.net/${type}/${id}`, {
        credentials: 'same-origin',
        signal: ctrl.signal,
      });
      if (res.status === 429 || res.status === 403) {
        penalize(lane, 15000);
        return null;
      }
      if (!res.ok) return null;
      const doc = new DOMParser().parseFromString(await res.text(), 'text/html');
      const txt = sel => (doc.querySelector(sel)?.textContent || '').trim();
      const num = s => parseFloat(s.replace(/,/g, ''));
      const ld = parseLd(doc);
      const score = ld ? ld.score : num(txt('[itemprop="ratingValue"]') || txt('.score-label'));
      const users = ld ? ld.scoredBy : num(txt('[itemprop="ratingCount"]'));
      const rank = num((txt('.ranked strong').match(/\d[\d,]*/) || [''])[0]);
      return {
        score: Number.isFinite(score) ? score : null,
        scoredBy: Number.isFinite(users) ? users : null,
        rank: Number.isFinite(rank) ? rank : null,
        title: txt('.title-name') || txt('h1'),
      };
    } catch (e) {
      log('MAL page fallback failed', e.name === 'AbortError' ? 'timeout' : e);
      return null;
    } finally {
      clearTimeout(timer);
    }
  }

  // ---------- rate limiting shared across all open MAL tabs ----------
  const lanes = [
    { name: 'jikan', gap: 1100, fails: 0, running: false, run: jikanEntry },
    { name: 'mal', gap: 450, fails: 0, running: false, run: scrapeMal },
  ];

  const slotKey = lane => `meta:next:${lane.name}`;

  // Claims the next free request slot for a source. The "next free time" lives in
  // shared storage, so several tabs together still respect each source's limit.
  async function reserveSlot(lane) {
    const now = Date.now();
    const next = Math.max(now, GM_getValue(slotKey(lane), 0));
    GM_setValue(slotKey(lane), next + lane.gap);
    if (next > now) await sleep(next - now);
  }

  // After a rate-limit/failure: pause the source for base * 2^fails ms (max 60s), in every tab.
  function penalize(lane, baseMs) {
    const ms = Math.min(60000, baseMs * (2 ** lane.fails));
    lane.fails++;
    GM_setValue(slotKey(lane), Math.max(GM_getValue(slotKey(lane), 0), Date.now() + ms));
    log(`${lane.name} backing off for ${Math.round(ms / 1000)}s`);
  }

  // ---------- job queue: visible links first, both sources pull from it ----------
  const MAX_ATTEMPTS = 2;
  const queue = [];
  const visible = new Map(); // "type:id" -> number of its links currently on screen

  function enqueue(type, id) {
    return new Promise(resolve => {
      queue.push({
        key: `${type}:${id}`, type, id, resolve, attempts: { jikan: 0, mal: 0 },
      });
      lanes.forEach(startLane);
    });
  }

  function takeJob(lane) {
    let idx = -1;
    for (let i = 0; i < queue.length; i++) {
      const j = queue[i];
      if (j.attempts[lane.name] >= MAX_ATTEMPTS) continue;
      if (visible.has(j.key)) { idx = i; break; } // on-screen link: take it now
      if (idx === -1) idx = i; // otherwise remember the oldest eligible job
    }
    return idx === -1 ? null : queue.splice(idx, 1)[0];
  }

  async function startLane(lane) {
    if (lane.running) return;
    lane.running = true;
    let job = takeJob(lane);
    while (job) {
      const cached = getEntry(job.key); // another tab may have fetched it meanwhile
      if (isFresh(cached)) {
        job.resolve({ ...cached, fromCache: true });
      } else {
        await reserveSlot(lane);
        job.attempts[lane.name]++;
        const entry = await lane.run(job.type, job.id, lane);
        if (entry && validScore(entry.score)) {
          lane.fails = 0;
          job.resolve(entry);
        } else if (entry && entry.score == null) {
          // A genuine no-score result is valid data, but do not let malformed values through.
          lane.fails = 0;
          job.resolve({ ...entry, score: null });
        } else if (Object.values(job.attempts).every(n => n >= MAX_ATTEMPTS)) {
          job.resolve(null);
        } else {
          queue.push(job); // let the other source (or a later retry) have a go
          lanes.forEach(startLane);
        }
      }
      job = takeJob(lane);
    }
    lane.running = false;
  }

  // ---------- fetching a single score ----------
  const inFlight = new Map();

  // cb(data, final): called with stale cached data first (final=false) if any,
  // then once more with the definitive result (final=true).
  function fetchScore(type, id, cb) {
    const key = `${type}:${id}`;
    const cached = getEntry(key);
    const bulk = getBulkEntry(type, id);

    if (isFresh(cached)) return cb(cached, true);
    // Always show the best known value immediately; stale values refresh quietly.
    const known = cached || (bulk ? { ...normaliseEntry(bulk), t: bulk.t || 0, source: 'bulk' } : null);
    if (known) cb(known, false);

    if (inFlight.has(key)) { inFlight.get(key).push(cb); return; }
    inFlight.set(key, [cb]);

    enqueue(type, id).then(entry => {
      const cbs = inFlight.get(key) || [];
      inFlight.delete(key);
      if (entry) {
        if (!entry.fromCache) setEntry(key, entry);
        cbs.forEach(f => f(entry, true));
      } else {
        // Preserve any known score on failure; only show N/A when nothing is known.
        cbs.forEach(f => f(known || { score: null, error: true }, true));
      }
    }).catch(error => {
      log('score job failed', key, error);
      const cbs = inFlight.get(key) || [];
      inFlight.delete(key);
      cbs.forEach(f => f(known || { score: null, error: true }, true));
    });
  }

  // ---------- badge rendering ----------
  // Every link gets one chip right after it; the score segment and the list segment live inside it.
  function getChip(link) {
    let chip = link._malChip;
    if (!chip) {
      chip = document.createElement('span');
      chip.className = 'mal-chip';
      link._malChip = chip;
      link.insertAdjacentElement('afterend', chip);
    }
    return chip;
  }

  function scoreTier(score) {
    if (score >= 8) return 't-great';
    if (score >= 7) return 't-good';
    if (score >= 6) return 't-mixed';
    return 't-poor';
  }

  function renderBadge(link, data) {
    let badge = link._malBadge;
    if (!badge) {
      badge = document.createElement('span');
      link._malBadge = badge;
      getChip(link).insertAdjacentElement('afterbegin', badge); // score segment is always first
    }

    if (data.loading) {
      badge.className = 'mal-seg mal-score loading';
      badge.textContent = '…';
      badge.title = 'Loading score…';
    } else if (validScore(data.score)) {
      badge.className = `mal-seg mal-score ${scoreTier(data.score)}`;
      badge.textContent = data.score.toFixed(2);
      const parts = [];
      if (data.scoredBy) parts.push(`${data.scoredBy.toLocaleString()} users`);
      if (data.rank) parts.push(`Rank #${data.rank.toLocaleString()}`);
      if (data.t) parts.push(`cached ${new Date(data.t).toLocaleDateString()}`);
      badge.title = parts.join(' • ');
    } else {
      badge.className = 'mal-seg mal-score na';
      badge.textContent = 'N/A';
      badge.title = data.error ? 'Could not load score (see console)' : 'No score yet';
    }
  }

  // ---------- settings ----------
  const SETTINGS_KEY = 'settings';
  const SEEN_MODES = ['dim', 'hide', 'off'];
  const settings = { seen: 'dim', manga: false, ...GM_getValue(SETTINGS_KEY, {}) };
  const saveSettings = () => GM_setValue(SETTINGS_KEY, settings);

  function toast(msg) {
    const t = document.createElement('div');
    t.textContent = msg;
    t.style.cssText = 'position:fixed;right:16px;bottom:16px;z-index:2147483647;background:#222;'
      + 'color:#fff;font:13px/1.3 -apple-system,"Segoe UI",Roboto,sans-serif;padding:8px 12px;'
      + 'border-radius:8px;box-shadow:0 2px 8px rgba(0,0,0,.4)';
    document.body.appendChild(t);
    setTimeout(() => t.remove(), 2500);
  }

  // ---------- your own anime list ----------
  // Imported from MAL's list endpoint (public lists, or your own while logged in)
  // and kept as one compact object: { user, t, items: { animeId: [status, score, watched, total] } }
  const LIST_KEY = 'list:anime';
  const LIST_REFRESH_MS = 24 * 60 * 60 * 1000;
  const STATUS_NAMES = { 1: 'Watching', 2: 'Completed', 3: 'On hold', 4: 'Dropped', 6: 'Plan to watch' };
  let myList = GM_getValue(LIST_KEY, null);
  let listBusy = false;

  function renderListBadge(link) {
    const info = link._malInfo;
    if (!info || info.type !== 'anime') return;
    const item = myList && myList.items[info.id];
    let badge = link._malListBadge;
    if (!item) {
      if (badge) {
        badge.remove();
        link._malListBadge = null;
      }
      return;
    }
    if (!badge) {
      badge = document.createElement('span');
      link._malListBadge = badge;
      getChip(link).insertAdjacentElement('beforeend', badge);
    }
    const [status, score, watched, total] = item;
    const progress = `${watched}/${total || '?'}`;
    const label = {
      1: `▶ ${progress}`, 2: score ? `✓ ${score}` : '✓', 3: `‖ ${progress}`, 4: `✗ ${progress}`, 6: 'PTW',
    }[status] || '?';
    badge.className = `mal-seg mal-list s${status}`;
    badge.textContent = status !== 2 && score ? `${label} · ${score}` : label;
    badge.title = [
      `On ${myList.user}'s list: ${STATUS_NAMES[status] || 'Unknown'}`,
      score ? `your score ${score}` : 'not scored',
      `${progress} episodes`,
    ].join(' • ');
  }

  // Anime already completed or dropped can be dimmed, or hidden when their list row holds only them.
  const SEEN_STATUSES = [2, 4];

  function countAnimeLinks(row) {
    let n = 0;
    row.querySelectorAll(LINK_SELECTOR).forEach(a => {
      const info = parseLink(a);
      if (info && info.type === 'anime') n++;
    });
    return n;
  }

  function clearSeen(link) {
    link.classList.remove('mal-seen');
    if (link._malChip) link._malChip.classList.remove('mal-seen');
    if (link._malHiddenRow) {
      link._malHiddenRow.classList.remove('mal-hide-seen');
      link._malHiddenRow = null;
    }
  }

  function applySeen(link) {
    clearSeen(link);
    const info = link._malInfo;
    if (!info || info.type !== 'anime' || settings.seen === 'off') return;
    if (location.pathname.startsWith('/animelist/')) return;
    const item = myList && myList.items[info.id];
    if (!item || !SEEN_STATUSES.includes(item[0])) return;
    const row = link.closest('li');
    if (settings.seen === 'hide' && row && countAnimeLinks(row) === 1) {
      row.classList.add('mal-hide-seen');
      link._malHiddenRow = row;
      return;
    }
    link.classList.add('mal-seen');
    if (link._malChip) link._malChip.classList.add('mal-seen');
  }

  function refreshListBadges() {
    document.querySelectorAll(LINK_SELECTOR).forEach(a => {
      if (!a._malInfo) return;
      renderListBadge(a);
      applySeen(a);
    });
  }

  async function fetchMyList(user) {
    const items = {};
    for (let offset = 0; ; offset += 300) {
      let rows = null;
      for (let tries = 0; tries < 3 && !rows; tries++) {
        await reserveSlot(lanes[1]);
        const res = await fetch(
          `https://myanimelist.net/animelist/${encodeURIComponent(user)}/load.json?offset=${offset}&status=7`,
          { credentials: 'same-origin' },
        );
        if (res.status === 429) {
          penalize(lanes[1], 15000);
          continue;
        }
        if (!res.ok) throw new Error(`HTTP ${res.status} (wrong username, or the list is private?)`);
        rows = await res.json();
      }
      if (!Array.isArray(rows)) throw new Error('unexpected response from MAL (rate limited?)');
      for (const r of rows) {
        items[r.anime_id] = [r.status, r.score, r.num_watched_episodes, r.anime_num_episodes];
      }
      if (rows.length < 300) break;
      await sleep(500);
    }
    return items;
  }

  async function importList(user, silent) {
    if (listBusy) return;
    listBusy = true;
    try {
      const items = await fetchMyList(user);
      myList = { user, t: Date.now(), items };
      GM_setValue(LIST_KEY, myList);
      refreshListBadges();
      const n = Object.keys(items).length;
      log(`imported ${n} anime from ${user}'s list`);
      if (!silent) alert(`Imported ${n} anime from ${user}'s list.`);
    } catch (e) {
      log('list import failed:', e.message);
      if (!silent) alert(`Could not import the list: ${e.message}`);
    } finally {
      listBusy = false;
    }
  }

  function parseUser(input) {
    const text = (input || '').trim();
    const m = text.match(/(?:animelist|mangalist|profile)\/([^/?#\s]+)/i);
    if (m) return decodeURIComponent(m[1]);
    return /^[\w-]{2,16}$/.test(text) ? text : null;
  }

  GM_registerMenuCommand('My list: set username or list URL…', () => {
    const input = prompt(
      'Your MAL username or anime list URL\n(e.g. https://myanimelist.net/animelist/LJEN94?status=7)',
      myList ? myList.user : '',
    );
    if (input === null) return;
    const user = parseUser(input);
    if (!user) {
      alert('Could not find a username in that.');
      return;
    }
    importList(user, false);
  });

  GM_registerMenuCommand('My list: refresh now', () => {
    if (!myList) {
      alert('Set your username first (My list: set username or list URL…).');
      return;
    }
    importList(myList.user, false);
  });

  GM_registerMenuCommand('My list: remove', () => {
    myList = null;
    GM_deleteValue(LIST_KEY);
    refreshListBadges();
  });

  GM_registerMenuCommand(`Seen anime (completed/dropped): ${settings.seen} - click to change`, () => {
    settings.seen = SEEN_MODES[(SEEN_MODES.indexOf(settings.seen) + 1) % SEEN_MODES.length];
    saveSettings();
    refreshListBadges();
    toast(`Seen anime: ${settings.seen}${myList ? '' : ' (import your list first)'}`);
  });

  GM_registerMenuCommand(`Manga links: ${settings.manga ? 'on' : 'off'} - click to toggle (reloads page)`, () => {
    settings.manga = !settings.manga;
    saveSettings();
    location.reload();
  });

  // Quietly refresh once a day; the shared lock stops several tabs doing it at once.
  setTimeout(() => {
    if (!myList || Date.now() - myList.t < LIST_REFRESH_MS) return;
    if (Date.now() - GM_getValue('meta:listLock', 0) < 2 * 60 * 1000) return;
    GM_setValue('meta:listLock', Date.now());
    importList(myList.user, true);
  }, 4000);

  // ---------- visibility tracking ----------
  function markVisible(link, on) {
    if (on === Boolean(link._malVisible)) return;
    link._malVisible = on;
    const n = (visible.get(link._malKey) || 0) + (on ? 1 : -1);
    if (n > 0) visible.set(link._malKey, n);
    else visible.delete(link._malKey);
  }

  function stopTracking(link) {
    markVisible(link, false);
    observer.unobserve(link);
  }

  function requestScore(link) {
    if (link._malRequested) return;
    link._malRequested = true;
    const { type, id } = link._malInfo;
    fetchScore(type, id, (data, final) => {
      renderBadge(link, data);
      if (final) stopTracking(link);
    });
  }

  const observer = new IntersectionObserver(entries => {
    entries.forEach(entry => {
      const link = entry.target;
      markVisible(link, entry.isIntersecting);
      if (entry.isIntersecting) requestScore(link);
    });
  }, { rootMargin: '600px' });

  // ---------- link detection (incremental: only newly added DOM is scanned) ----------
  function parseLink(a) {
    let url;
    try { url = new URL(a.href, location.href); } catch (e) { return null; }
    if (url.hostname !== 'myanimelist.net') return null;
    const m = url.pathname.match(LINK_RE);
    if (!m || (m[1] === 'manga' && !settings.manga)) return null;
    return { type: m[1], id: m[2] };
  }

  const seen = new WeakSet();

  function processLink(a) {
    if (seen.has(a)) return;
    seen.add(a);
    if (!a.textContent.trim()) return;
    if (a.closest('#headerSmall, .header-menu, #footer, .horiznav_nav')) return;
    const info = parseLink(a);
    if (!info) return;
    if (a.nextElementSibling?.classList.contains('mal-chip')) return;

    a._malInfo = info;
    a._malKey = `${info.type}:${info.id}`;
    const cached = getEntry(a._malKey);
    const bulk = getBulkEntry(info.type, info.id);
    const known = cached || (bulk ? { ...normaliseEntry(bulk), t: bulk.t || 0, source: 'bulk' } : null);
    if (isFresh(cached)) {
      renderBadge(a, cached); // live cache is authoritative when fresh
    } else if (known) {
      renderBadge(a, known); // stale cache or bulk data appears immediately
      observer.observe(a); // refresh in background when near the viewport
    } else {
      renderBadge(a, { loading: true });
      observer.observe(a); // fetched once it is near the viewport
    }
    renderListBadge(a);
    applySeen(a);
  }

  function scanRoot(root) {
    if (root.matches?.(LINK_SELECTOR)) processLink(root);
    if (root.querySelectorAll) root.querySelectorAll(LINK_SELECTOR).forEach(processLink);
  }

  scanRoot(document.body);

  const pendingRoots = new Set();
  let scanTimer;
  new MutationObserver(mutations => {
    for (const m of mutations) {
      for (const n of m.addedNodes) {
        if (n.nodeType === 1 && !n.classList.contains('mal-chip')) pendingRoots.add(n);
      }
    }
    if (!pendingRoots.size) return;
    clearTimeout(scanTimer);
    scanTimer = setTimeout(() => {
      const roots = [...pendingRoots];
      pendingRoots.clear();
      roots.forEach(scanRoot);
    }, 300);
  }).observe(document.body, { childList: true, subtree: true });

  GM_registerMenuCommand('Cache: import from JSON file', () => {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = 'application/json';
    input.onchange = () => {
      const file = input.files[0];
      if (!file) return;
      file.text().then(txt => {
        try {
          const obj = JSON.parse(txt);
          let n = 0;
          for (const [k, v] of Object.entries(obj)) {
            if (/^(anime|manga):\d+$/.test(k) && v && typeof v === 'object') {
              GM_setValue(PREFIX + k, v);
              n++;
            }
          }
          alert(`Imported ${n} entries.`);
        } catch (e) { alert('Invalid JSON file.'); }
      });
    };
    input.click();
  });

  // ---------- pre-populated bulk score database ----------
  // Accepted JSON formats:
  // { "anime:5114": { "score": 9.1, "scoredBy": 1000, "rank": 1, "title": "..." } }
  // { "data": [{ "mal_id": 5114, "score": 9.1, "scored_by": 1000, "rank": 1, "title": "..." }] }
  GM_registerMenuCommand('Bulk database: import JSON…', () => {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = 'application/json,.json';
    input.onchange = () => {
      const file = input.files && input.files[0];
      if (!file) return;
      file.text().then(txt => {
        try {
          const parsed = JSON.parse(txt);
          const source = Array.isArray(parsed) ? parsed : (Array.isArray(parsed.data) ? parsed.data : null);
          const entries = [];
          if (source) {
            source.forEach(row => {
              const id = Number(row.mal_id ?? row.id);
              if (Number.isSafeInteger(id) && id > 0 && row && typeof row === 'object') {
                entries.push([String(id), normaliseEntry(row)]);
              }
            });
          } else if (parsed && typeof parsed === 'object') {
            Object.entries(parsed).forEach(([key, row]) => {
              const match = key.match(/^(?:anime:)?(\d+)$/);
              if (match && row && typeof row === 'object') entries.push([match[1], normaliseEntry(row)]);
            });
          }
          let imported = 0;
          entries.forEach(([id, row]) => {
            if (!validScore(row.score)) return;
            GM_setValue(BULK_PREFIX + id, { ...row, t: Date.now() });
            imported++;
          });
          alert(`Imported ${imported.toLocaleString()} valid anime scores into the separate bulk database.`);
          log(`bulk database imported ${imported} scores`);
        } catch (e) {
          alert('Could not import JSON: invalid JSON or unsupported format.');
          log('bulk import failed', e);
        }
      });
    };
    input.click();
  });

  // ---------- bulk preload (25 anime per request, shares the Jikan rate limit) ----------
  async function preloadList(basePath, maxPages) {
    let saved = 0;
    for (let p = 1; p <= maxPages; p++) {
      await reserveSlot(lanes[0]);
      const json = await jikan(`${basePath}?page=${p}&limit=25`);
      if (!json || !json.data) break;
      for (const d of json.data) {
        setEntry(`anime:${d.mal_id}`, toEntry(d));
        saved++;
      }
      log(`preload ${basePath} page ${p}/${maxPages} (${saved} saved)`);
      if (!json.pagination?.has_next_page) break;
    }
    return saved;
  }

  GM_registerMenuCommand('Preload: top anime…', async () => {
    const n = parseInt(prompt('How many top-ranked anime to preload? (25 per request, ~1 request/sec)', '500'), 10);
    if (!n || n < 1) return;
    const saved = await preloadList('/top/anime', Math.ceil(n / 25));
    alert(`Preloaded ${saved} anime scores.`);
  });

  GM_registerMenuCommand('Preload: currently airing season', async () => {
    const saved = await preloadList('/seasons/now', 20);
    alert(`Preloaded ${saved} airing anime.`);
  });

  GM_registerMenuCommand('Preload: upcoming season', async () => {
    const saved = await preloadList('/seasons/upcoming', 20);
    alert(`Preloaded ${saved} upcoming anime.`);
  });
